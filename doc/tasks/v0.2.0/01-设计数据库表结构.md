## 任务描述

创建 data-cache 模块，设计 SQLite 数据库表结构，覆盖物品、技能、天赋、词缀等核心数据。

## 详细步骤

### 1. 创建 `data-cache/build.gradle.kts`

```kotlin
dependencies {
    implementation(project(":common"))
    implementation("org.xerial:sqlite-jdbc:3.44.1.0")
}
```

### 2. 建表 SQL（`migration/v001_init.sql`）

```sql
-- 基础物品表
CREATE TABLE IF NOT EXISTS base_items (
    id          INTEGER PRIMARY KEY,
    name        TEXT NOT NULL,
    name_zh     TEXT,
    class       TEXT NOT NULL,
    inventory_width  INTEGER DEFAULT 0,
    inventory_height INTEGER DEFAULT 0,
    requirements TEXT,       -- JSON
    implicits   TEXT,        -- JSON
    properties  TEXT,        -- JSON
    flavour_text TEXT,
    drop_level  INTEGER DEFAULT 0,
    wiki_url    TEXT,
    version     TEXT NOT NULL
);
CREATE INDEX idx_base_items_name ON base_items(name);
CREATE INDEX idx_base_items_class ON base_items(class);

-- 技能宝石表
CREATE TABLE IF NOT EXISTS skill_gems (
    id            INTEGER PRIMARY KEY,
    name          TEXT NOT NULL,
    name_zh       TEXT,
    gem_type      TEXT NOT NULL,  -- 'active'/'support'
    gem_tags      TEXT,           -- JSON array
    primary_attribute TEXT,       -- 'str'/'dex'/'int'
    description   TEXT,
    quality_stats TEXT,
    level_stats   TEXT,
    required_level INTEGER DEFAULT 1,
    version       TEXT NOT NULL
);
CREATE INDEX idx_skill_gems_name ON skill_gems(name);

-- 天赋点表
CREATE TABLE IF NOT EXISTS passive_skills (
    id          INTEGER PRIMARY KEY,
    name        TEXT NOT NULL,
    name_zh     TEXT,
    class       TEXT,
    ascendancy  TEXT,
    stats       TEXT,        -- JSON
    is_keystone INTEGER DEFAULT 0,
    is_notable  INTEGER DEFAULT 0,
    is_jewel_socket INTEGER DEFAULT 0,
    x           REAL,
    y           REAL,
    connections TEXT,        -- JSON array
    version     TEXT NOT NULL
);

-- 词缀表
CREATE TABLE IF NOT EXISTS mods (
    id              INTEGER PRIMARY KEY,
    name            TEXT NOT NULL,
    name_zh         TEXT,
    mod_type        TEXT,   -- 'prefix'/'suffix'/'implicit'/'enchant'
    domain          TEXT,
    generation_type TEXT,
    mod_group       TEXT,
    stats           TEXT,   -- JSON
    spawn_tags      TEXT,   -- JSON
    spawn_weights   TEXT,   -- JSON
    required_level  INTEGER DEFAULT 0,
    version         TEXT NOT NULL
);

-- 数据版本表
CREATE TABLE IF NOT EXISTS data_version (
    table_name   TEXT PRIMARY KEY,
    last_sync    TEXT,
    record_count INTEGER DEFAULT 0,
    wiki_version TEXT
);

-- 翻译映射表
CREATE TABLE IF NOT EXISTS translations (
    source  TEXT NOT NULL,
    target  TEXT NOT NULL,
    domain  TEXT NOT NULL,  -- 'item'/'skill'/'passive'/'mod'/'map'
    PRIMARY KEY (source, domain)
);

-- FTS5 全文搜索
CREATE VIRTUAL TABLE IF NOT EXISTS items_fts USING fts5(
    name, name_zh, class, 
    content='base_items', content_rowid='id'
);
```
