```java

    String version;
    Map<Integer, PassiveSkill> nodeMap;     // id → node
    Map<Integer, List<Integer>> graph;      // 邻接表（id → 相连节点列表）
    double minX, maxX, minY, maxY;          // 边界
    
    // 分类索引
    Map<String, List<Integer>> keystoneNodes;   // 核心天赋
    Map<String, List<Integer>> notableNodes;    // 显著天赋
    Map<String, List<Integer>> jewelSockets;    // 珠宝孔
    Map<String, List<Integer>> ascendancyStart; // 升华起点
    Map<String, Map<String, List<Integer>>> classStarts; // 各职业起点
}
```
