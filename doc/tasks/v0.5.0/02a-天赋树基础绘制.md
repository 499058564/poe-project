## 详细步骤

### 1. PassiveTreeCanvas 组件框架

```java
public class PassiveTreeCanvas extends Canvas {
    private PassiveTree tree;
    
    // 坐标变换
    private double scale = 1.0;
    private double offsetX, offsetY;
    
    // 视口裁剪
    private double viewportMinX, viewportMinY;
    private double viewportMaxX, viewportMaxY;
    
    // 节点状态
    private Set<Integer> allocatedNodes = new HashSet<>();
    private Integer hoveredNodeId;
    private Integer selectedNodeId;
    
    /** 世界坐标 → 屏幕坐标 */
    double worldToScreenX(double wx) { return (wx - offsetX) * scale + getWidth() / 2; }
    double worldToScreenY(double wy) { return (wy - offsetY) * scale + getHeight() / 2; }
    
    /** 屏幕坐标 → 世界坐标 */
    double screenToWorldX(double sx) { return (sx - getWidth() / 2) / scale + offsetX; }
    double screenToWorldY(double sy) { return (sy - getHeight() / 2) / scale + offsetY; }
    
    /** 初始居中显示天赋树 */
    public void centerTree() {
        offsetX = (tree.getMinX() + tree.getMaxX()) / 2;
        offsetY = (tree.getMinY() + tree.getMaxY()) / 2;
        scale = Math.min(getWidth() / tree.getWidth(), getHeight() / tree.getHeight()) * 0.8;
        render();
    }
}
```

### 2. 连线绘制

```java
private void drawConnections(GraphicsContext gc) {
    gc.setStroke(Color.rgb(60, 60, 40));
    gc.setLineWidth(Math.max(1.0, 1.5 / scale));

    for (var entry : tree.getGraph().entrySet()) {
        PassiveSkill from = tree.getNode(entry.getKey());
        for (int toId : entry.getValue()) {
            PassiveSkill to = tree.getNode(toId);
            
            double x1 = worldToScreenX(from.getX());
            double y1 = worldToScreenY(from.getY());
            double x2 = worldToScreenX(to.getX());
            double y2 = worldToScreenY(to.getY());
            
            if (!isLineVisible(x1, y1, x2, y2)) continue;
            gc.strokeLine(x1, y1, x2, y2);
        }
    }
}
```

### 4. 渲染主流程

```java
private void render() {
    GraphicsContext gc = getGraphicsContext2D();
    gc.clearRect(0, 0, getWidth(), getHeight());
    
    updateViewport();        // 计算视口范围
    drawConnections(gc);     // 连线
    drawNodes(gc);           // 节点
}

private void updateViewport() {
    viewportMinX = screenToWorldX(-50);
    viewportMinY = screenToWorldY(-50);
    viewportMaxX = screenToWorldX(getWidth() + 50);
    viewportMaxY = screenToWorldY(getHeight() + 50);
}

private boolean isInViewport(double sx, double sy, double margin) {
    return sx >= -margin && sx <= getWidth() + margin
        && sy >= -margin && sy <= getHeight() + margin;
}
```
