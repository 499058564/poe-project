### 3. PobProcessManager — 进程管理

```java
public class PobProcessManager {
    private Process luaProcess;
    private BufferedReader reader;
    private BufferedWriter writer;

    /** 启动子进程 */
    public void start() throws IOException {
        Path pobPath = new PobPathResolver().resolve()
            .orElseThrow(() -> new PoeException("未找到 POB 安装"));

        ProcessBuilder pb = new ProcessBuilder(
            pobPath.resolve("lua/lua.exe").toString(),
            pobPath.resolve("launch.lua").toString(),
            "--headless"
        );
        pb.redirectErrorStream(true);
        luaProcess = pb.start();

        reader = new BufferedReader(new InputStreamReader(luaProcess.getInputStream()));
        writer = new BufferedWriter(new OutputStreamWriter(luaProcess.getOutputStream()));

        waitForReady();  // 读取就绪信号
    }

    /** 等待就绪信号（超时 10s） */
    private void waitForReady() { ... }

    /** 健康检查：进程是否存活 */
    public boolean isRunning() {
        return luaProcess != null && luaProcess.isAlive();
    }

    /** 优雅关闭 */
    public void stop() {
        sendRaw("{\"action\":\"shutdown\"}");
        luaProcess.waitFor(5, TimeUnit.SECONDS);
        luaProcess.destroyForcibly();
    }
```

### 4. JsonProtocol — 请求-响应协议

```java
public class JsonProtocol {
    private final PobProcessManager processManager;
    private final Map<String, CompletableFuture<JsonNode>> pendingRequests;
    private final ScheduledExecutorService timeoutChecker;

    /** 发送请求并异步等待响应 */
    public CompletableFuture<JsonNode> sendRequest(String action, JsonNode payload) {
        String requestId = UUID.randomUUID().toString();

        JsonNode request = mapper.createObjectNode()
            .put("id", requestId)
            .put("action", action)
            .set("payload", payload);

        CompletableFuture<JsonNode> future = new CompletableFuture<>();
        pendingRequests.put(requestId, future);

        processManager.sendRaw(request.toString());

        // 超时处理
        timeoutChecker.schedule(() -> {
            future.completeExceptionally(new TimeoutException("请求超时: " + action));
            pendingRequests.remove(requestId);
        }, 30, TimeUnit.SECONDS);

        return future;
    }

    /** 后台线程持续读取响应并匹配 requestId */
    public void startResponseListener() { ... }
}
```

### 5. 协议格式

```json
// 请求
{"id":"abc","action":"calculate","payload":{"build":"<xml>...</xml>"}}

// 成功响应
{"id":"abc","action":"calculate","success":true,"data":{...},"error":null}

// 错误响应
{"id":"abc","action":"calculate","success":false,"data":null,"error":"message"}
```
